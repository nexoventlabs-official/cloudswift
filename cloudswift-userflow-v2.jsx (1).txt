function FlowNode({ node, active, onClick, color }) {
  const c = CHANNEL_COLORS[color || node.color || node.type || "neutral"];
  const isActive = active === node.id;
  return (
    <div
      onClick={() => onClick(node.id)}
      style={{
        border: `1px solid ${isActive ? c.border : "#E5E7EB"}`,
        borderRadius: 10,
        padding: "10px 14px",
        cursor: "pointer",
        background: isActive ? c.bg : "#FAFAFA",
        transition: "all 0.15s",
        position: "relative",
        minWidth: 140,
      }}
    >
      <div style={{ display: "flex", alignItems: "center", gap: 6, marginBottom: 3 }}>
        <div style={{ width: 7, height: 7, borderRadius: "50%", background: c.dot, flexShrink: 0 }} />
        <div style={{ fontWeight: 600, fontSize: 13, color: isActive ? c.text : "#111" }}>{node.label}</div>
      </div>
      <div style={{ fontSize: 11, color: "#6B7280", paddingLeft: 13 }}>{node.sub}</div>
      {node.type === "wa" && (
        <div style={{
          position: "absolute", top: 8, right: 8,
          fontSize: 10, background: "#DCF8C6", color: "#128C7E",
          borderRadius: 4, padding: "1px 6px", fontWeight: 600
        }}>WA</div>
      )}
    </div>
  );
}

                )}

                {/* WA message */}
                {PATHS[active]?.message && (
                  <div style={{ marginBottom: 14 }}>
                    <div style={{ fontSize: 11, fontFamily: "monospace", letterSpacing: "0.1em", textTransform: "uppercase", color: "#9CA3AF", marginBottom: 8 }}>
                      WhatsApp message
                    </div>
                    <WaMessage msg={PATHS[active].message} />
                  </div>
                )}

                {/* No message */}
                {!PATHS[active]?.message && PATHS[active] && (
                  <div style={{
                    background: "#F9FAFB",
                    borderRadius: 8,
                    padding: "12px 14px",
                    fontSize: 13,
                    color: "#6B7280",
                    lineHeight: 1.6
                  }}>
                    {active === "qualify_score" && "The system automatically scores based on company size, intent, and timeline. No human review needed at this stage."}
                    {active === "hot" && "Routes immediately to sales brief. Company size 500+ always triggers hot regardless of timeline — enterprise leads in research phase are still priority."}
                    {active === "warm" && "Enters 21-day nurture sequence. Day 3 / 7 / 21. Sales rep notified automatically if prospect engages with Day 7 message."}
                    {active === "cold" && "Exits cleanly with a referral ask and free assessment offer. No follow-up sequence. Re-enters warm flow if they engage with the assessment."}
                    {active === "website_landing" && "Prototype is live at cloudswift-demo.vercel.app. Needs DNS switch to oncloudswift.com before any paid traffic is sent."}
                    {active === "website_cta" && 'CTA button opens wa.me link directly. Pre-fills "Hi, I\'d like to know more about CloudSwift" so the prospect doesn\'t have to type anything.'}
                    {active === "calendly" && "30-minute discovery call. Calendly link sent inline in WhatsApp — no email needed. Slot confirmation triggers pre-call brief to sales rep."}
                    {active === "discovery_call" && "Sales rep runs the call with the pre-call brief in hand. She knows the company size, current setup, pain point, and timeline before saying hello."}
                    {active === "proposal" && "Sales rep sends formal scoped proposal within 48 hours of discovery call. Pricing anchored to engagement type."}
                    {active === "closed_won" && "Client onboarding begins. They enter a separate existing-client WhatsApp flow for support triage, renewal nudges, and upsell sequences."}
                    {active === "nurture_reentry" && 'Any positive reply at any point in the nurture sequence — including "yes", "interested", "tell me more" — automatically triggers the hot lead flow and notifies sales rep within 2 minutes.'}
                    {["google_ad", "meta_ad", "website_organic", "cold_email", "referral"].includes(active) && "Entry point. Tap a downstream node to see the message or action that follows."}
                  </div>
                )}

                {/* Next step */}
                {PATHS[active]?.next && (
                  <div style={{ marginTop: 14, paddingTop: 12, borderTop: "1px solid #F3F4F6" }}>
                    <div style={{ fontSize: 11, fontFamily: "monospace", letterSpacing: "0.1em", textTransform: "uppercase", color: "#9CA3AF", marginBottom: 6 }}>
                      Next step
                    </div>
                    <button
                      onClick={() => handleClick(PATHS[active].next)}
                      style={{
                        border: "1px solid #E5E7EB",
                        borderRadius: 8,
                        background: "#F9FAFB",
                        cursor: "pointer",
                        padding: "10px 14px",
                        fontSize: 13,
                        color: "#111",
                        fontWeight: 500,
                        width: "100%",
                        textAlign: "left",
                        display: "flex",
                        justifyContent: "space-between",
                        alignItems: "center"
                      }}
                    >
                      {PATHS[PATHS[active].next]?.label || PATHS[active].next}
                      <span style={{ color: "#9CA3AF" }}>→</span>
                    </button>
                  </div>
                )}
              </div>
            </div>
          )}

          {/* Legend */}
          <div style={{
            background: "#fff",
            borderRadius: 12,
            border: "1px solid #E5E7EB",
            padding: "14px 16px",
            marginTop: 12
          }}>
            <div style={{ fontSize: 10, fontFamily: "monospace", letterSpacing: "0.12em", textTransform: "uppercase", color: "#9CA3AF", marginBottom: 10 }}>
              Legend
            </div>
            {[
              { dot: "#4285F4", label: "Google Ads path" },
              { dot: "#6366F1", label: "Meta CTWA path" },
              { dot: "#25D366", label: "WhatsApp flow" },
              { dot: "#EF4444", label: "Hot lead path" },
              { dot: "#F59E0B", label: "Warm / nurture" },
              { dot: "#9CA3AF", label: "Cold / exit" },
              { dot: "#16A34A", label: "Closed won" },
            ].map(({ dot, label }) => (
              <div key={label} style={{ display: "flex", alignItems: "center", gap: 8, padding: "4px 0", fontSize: 12, color: "#374151" }}>
                <div style={{ width: 8, height: 8, borderRadius: "50%", background: dot, flexShrink: 0 }} />
                {label}
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}
